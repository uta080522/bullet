cmake_minimum_required(VERSION 3.24)
project(bullet VERSION 0.1.0 LANGUAGES C)

set(CMAKE_C_STANDARD 99)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
if(MSVC)
  set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
endif()

find_package(SDL3 3.2 CONFIG REQUIRED)
# Use WebP's config, not SDL_image 3.4.4's mis-cased Findwebp module.
find_package(WebP CONFIG REQUIRED)
find_package(SDL3_image 3.4 CONFIG REQUIRED)
find_package(SDL3_ttf CONFIG REQUIRED)
find_package(cJSON CONFIG REQUIRED)
find_package(CURL REQUIRED)
if(NOT WIN32)
  find_package(OpenSSL REQUIRED COMPONENTS Crypto)
endif()

option(SANITIZE "Enable memory/undefined-behavior checks" OFF)
include(CTest)
add_executable(bullet bullet.c)
set(programs bullet)
if(BUILD_TESTING)
  add_executable(bullet-test test.c)
  list(APPEND programs bullet-test)
  add_test(NAME unit COMMAND bullet-test)
  add_test(NAME cli COMMAND bullet-test --cli "$<TARGET_FILE:bullet>")
  if(SANITIZE)
    set_tests_properties(unit cli PROPERTIES ENVIRONMENT
      "ASAN_OPTIONS=halt_on_error=1:exitcode=86;UBSAN_OPTIONS=halt_on_error=1:exitcode=87")
  endif()
endif()

foreach(program IN LISTS programs)
  target_link_libraries(${program} PRIVATE SDL3::SDL3
    SDL3_image::SDL3_image SDL3_ttf::SDL3_ttf cjson CURL::libcurl)
  if(WIN32)
    target_link_libraries(${program} PRIVATE bcrypt crypt32)
  else()
    target_link_libraries(${program} PRIVATE OpenSSL::Crypto m)
    target_compile_definitions(${program} PRIVATE _POSIX_C_SOURCE=200809L)
  endif()
  if(MSVC)
    target_compile_definitions(${program} PRIVATE _CRT_SECURE_NO_WARNINGS)
    target_compile_options(${program} PRIVATE /W4 /WX /utf-8)
    if(SANITIZE)
      target_compile_options(${program} PRIVATE /fsanitize=address /Zi)
      target_link_options(${program} PRIVATE /DEBUG /INCREMENTAL:NO)
    endif()
  else()
    target_compile_options(${program} PRIVATE
      -Wall -Wextra -Wpedantic -Werror)
    if(SANITIZE)
      target_compile_options(${program} PRIVATE
        -fsanitize=address,undefined -fno-omit-frame-pointer)
      target_link_options(${program} PRIVATE -fsanitize=address,undefined)
    endif()
  endif()
endforeach()
